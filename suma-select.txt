      
import { Component } from '@angular/core';
@Component({
  selector: 'app-servicios',
  templateUrl: './servicios.component.html'
})
export class ServiciosComponent {
  listaServicios = [
    { id: 1, nombre: 'Corte básico', precio: 10 },
    { id: 2, nombre: 'Tinte', precio: 15 },
    { id: 3, nombre: 'Peinado', precio: 8 }
  ];

  serviciosSeleccionados: any[] = [];
  totalSeleccionado = 0;

  calcularTotal(): void {
    this.totalSeleccionado = this.serviciosSeleccionados
      .reduce((acc, s) => acc + s.precio, 0);
  }
}

<div class="mb-3">
  <label class="form-label fw-bold">Servicios disponibles</label>
  <select class="form-select" multiple [(ngModel)]="serviciosSeleccionados" (change)="calcularTotal()">
    <option *ngFor="let servicio of listaServicios" [ngValue]="servicio">
      {{ servicio.nombre }} - {{ servicio.precio | currency:'EUR':'symbol':'1.0-0' }}
    </option>
  </select>
</div>

<!-- Mostrar el total fuera del select -->
<div class="mt-3">
  <h5>Total: {{ totalSeleccionado | currency:'EUR':'symbol':'1.0-0' }}</h5>
</div>
